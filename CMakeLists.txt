cmake_minimum_required(VERSION 3.24)
project(JacobPlatform VERSION 0.1 LANGUAGES CXX)

# Independent of the legacy DX11/FBX editor.
if(WIN32)
    set(default_backend WIN32)
else()
    set(default_backend SDL)
endif()
set(JACOB_PLATFORM_BACKEND "${default_backend}" CACHE STRING "Window implementation: WIN32 or SDL")
set_property(CACHE JACOB_PLATFORM_BACKEND PROPERTY STRINGS WIN32 SDL)
option(JACOB_FETCH_SDL "Fetch the pinned SDL source when building the SDL backend" ON)
option(JACOB_HEADLESS_TESTS "Run SDL tests with the dummy driver (no display/GPU validation)" OFF)
option(JACOB_SANITIZERS "Enable address/undefined sanitizers for our platform code" OFF)

add_library(JacobPlatform STATIC
    Engine/CoreEngine/Platform/ResourcePaths.cpp
    Engine/CoreEngine/Core/Window/Window.cpp)
target_compile_features(JacobPlatform PUBLIC cxx_std_20)
target_include_directories(JacobPlatform PUBLIC Engine/CoreEngine)

if(JACOB_PLATFORM_BACKEND STREQUAL "WIN32")
    if(NOT WIN32)
        message(FATAL_ERROR "The WIN32 backend requires a Windows toolchain.")
    endif()
    target_sources(JacobPlatform PRIVATE Engine/CoreEngine/Platform/Win32/WindowWin32.cpp)
    target_link_libraries(JacobPlatform PRIVATE user32)
    target_compile_definitions(JacobPlatform PUBLIC JACOB_PLATFORM_WIN32=1)
elseif(JACOB_PLATFORM_BACKEND STREQUAL "SDL")
    if(JACOB_FETCH_SDL)
        include(FetchContent)
        set(SDL_SHARED OFF CACHE BOOL "" FORCE)
        set(SDL_STATIC ON CACHE BOOL "" FORCE)
        set(SDL_TEST_LIBRARY OFF CACHE BOOL "" FORCE)
        set(SDL_TESTS OFF CACHE BOOL "" FORCE)
        set(SDL_EXAMPLES OFF CACHE BOOL "" FORCE)
        FetchContent_Declare(SDL3
            URL https://codeload.github.com/libsdl-org/SDL/tar.gz/7f3ae3d57459e59943a4ecfefc8f6277ec6bf540
            URL_HASH SHA256=7a8347c770b90b33daac2352858ca03f9c9a2ccc8ce711054870361d1a6b32e5
            DOWNLOAD_EXTRACT_TIMESTAMP TRUE) # release-3.2.28
        FetchContent_MakeAvailable(SDL3)
    else()
        find_package(SDL3 3.2.28 CONFIG REQUIRED)
    endif()
    target_sources(JacobPlatform PRIVATE Engine/CoreEngine/Platform/SDL/WindowSDL.cpp)
    target_link_libraries(JacobPlatform PUBLIC SDL3::SDL3)
    target_compile_definitions(JacobPlatform PUBLIC JACOB_PLATFORM_SDL=1)
else()
    message(FATAL_ERROR "Unknown JACOB_PLATFORM_BACKEND: ${JACOB_PLATFORM_BACKEND}")
endif()

function(jacob_checks target)
    if(MSVC)
        target_compile_options(${target} PRIVATE /W4 /permissive- /utf-8)
    else()
        target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Werror)
        if(JACOB_SANITIZERS)
            target_compile_options(${target} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
            target_link_options(${target} PRIVATE -fsanitize=address,undefined)
        endif()
    endif()
endfunction()
jacob_checks(JacobPlatform)

if(JACOB_PLATFORM_BACKEND STREQUAL "WIN32")
    # Compile the real legacy-editor adapter too, without requiring FBX or D3D.
    add_library(JacobImGuiWin32Bridge OBJECT Engine/CoreEngine/GUI/Platform/ImGuiWindowWin32.cpp)
    target_link_libraries(JacobImGuiWin32Bridge PRIVATE JacobPlatform)
    target_include_directories(JacobImGuiWin32Bridge PRIVATE Engine/CoreEngine/Vender/include)
    jacob_checks(JacobImGuiWin32Bridge)
endif()

add_executable(PlatformSmoke tools/PlatformSmoke.cpp)
target_link_libraries(PlatformSmoke PRIVATE JacobPlatform)
jacob_checks(PlatformSmoke)

include(CTest)
if(BUILD_TESTING)
    if(JACOB_PLATFORM_BACKEND STREQUAL "WIN32")
        set(imgui_dir Engine/CoreEngine/Vender/include/imgui)
        add_library(JacobImGuiTestSupport STATIC
            ${imgui_dir}/imgui.cpp ${imgui_dir}/imgui_draw.cpp
            ${imgui_dir}/imgui_widgets.cpp ${imgui_dir}/imgui_tables.cpp
            ${imgui_dir}/imgui_impl_win32.cpp)
        target_include_directories(JacobImGuiTestSupport PUBLIC Engine/CoreEngine/Vender/include)
        target_link_libraries(JacobImGuiTestSupport PRIVATE user32 gdi32 dwmapi imm32)
        add_executable(ImGuiBridgeTests tests/platform/ImGuiBridgeTests.cpp)
        target_link_libraries(ImGuiBridgeTests PRIVATE JacobPlatform JacobImGuiWin32Bridge JacobImGuiTestSupport)
        jacob_checks(ImGuiBridgeTests)
        add_test(NAME platform.imgui_bridge COMMAND ImGuiBridgeTests)
        set_tests_properties(platform.imgui_bridge PROPERTIES TIMEOUT 30)
    endif()
    add_executable(PlatformTests tests/platform/PlatformTests.cpp)
    target_link_libraries(PlatformTests PRIVATE JacobPlatform)
    jacob_checks(PlatformTests)
    add_test(NAME platform.paths COMMAND PlatformTests paths)
    add_test(NAME platform.input COMMAND PlatformTests input)
    add_test(NAME platform.window COMMAND PlatformTests window)
    add_test(NAME platform.facade COMMAND PlatformTests facade)
    add_test(NAME platform.smoke COMMAND PlatformSmoke --smoke-test)
    set_tests_properties(platform.paths platform.input platform.window platform.facade platform.smoke PROPERTIES TIMEOUT 30)
    if(JACOB_HEADLESS_TESTS)
        if(NOT JACOB_PLATFORM_BACKEND STREQUAL "SDL")
            message(FATAL_ERROR "Headless tests require the SDL backend.")
        endif()
        set_tests_properties(platform.window platform.facade platform.smoke PROPERTIES ENVIRONMENT "SDL_VIDEODRIVER=dummy")
    endif()
endif()
